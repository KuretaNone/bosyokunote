```ts
import { DOMParser } from "jsr:@b-fuze/deno-dom";

async function scrape() {
    const res = await fetch("https://www.shonenjump.com/j/weeklyshonenjump/");
    const source = await res.text();

    const DOM = new DOMParser().parseFromString(source, "text/html");

    const items = DOM.querySelectorAll('.s_jump li > p:first-of-type');
    const Target = DOM.querySelectorAll('[class="ttl"]');

    const Results : string[] = [];

    items.forEach(el => {
        const text = el.textContent?.trim();

        if (text?.startsWith("週刊少年ジャンプ")) {
            console.log(text);
            Results.push(text);
        }
    });

    Target.forEach(el => {
        const text = el.innerText;
        Results.push(text);
        console.log(text);
    });
}
    
scrape();
```
